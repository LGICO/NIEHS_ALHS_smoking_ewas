awk -F '\t' '
  BEGIN {OFS=FS}

  NR==FNR {
    line=$0
    while (match(line, /"[^"]+"/)) {
      id=substr(line, RSTART+1, RLENGTH-2)
      gsub(/_DNA_/, "-DNA_", id)
      wanted[id]=1
      line=substr(line, RSTART+RLENGTH)
    }
    next
  }

  FNR==1 {print; next}

  {
    id=$1
    gsub(/_DNA_/, "-DNA_", id)
  }

  $17!="PASS" && (id in wanted) {print}
' sample_list.txt \
  identity_results_filtered095/best_matches.tsv \
  > identity_results_filtered095/overlapping_bad_samples.tsv
